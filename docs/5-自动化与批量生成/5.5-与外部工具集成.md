## 功能描述

外部工具集成是指将 ComfyUI 与其他软件、服务或平台连接起来，构建更强大的图像生成系统。这就像是给 ComfyUI 装上了"扩展接口"，可以与 Web 应用、数据库、自动化工具等协同工作。

### 集成原理

ComfyUI 与外部工具集成主要通过以下方式实现：

1. **REST API**：通过 HTTP 请求与 ComfyUI 通信
2. **WebSocket**：实时接收生成进度和状态
3. **文件系统**：通过输入/输出目录交换文件
4. **消息队列**：使用 Redis、RabbitMQ 等实现异步任务处理

## 5.5.1 REST API 封装示例

### 功能描述

使用 Flask 或 FastAPI 封装 ComfyUI 的 API，提供更简洁、更安全的接口给外部应用调用。

### Flask 封装示例

```python
from flask import Flask, request, jsonify
import requests
import json
import os
import uuid

app = Flask(__name__)
COMFYUI_SERVER = "http://127.0.0.1:8188"

@app.route('/api/generate', methods=['POST'])
def generate():
    """图像生成接口"""
    data = request.json
    
    with open('templates/basic_txt2img.json') as f:
        workflow = json.load(f)
    
    workflow["6"]["inputs"]["text"] = data.get('prompt', '')
    workflow["7"]["inputs"]["text"] = data.get('negative_prompt', '')
    workflow["3"]["inputs"]["seed"] = data.get('seed', -1)
    workflow["3"]["inputs"]["steps"] = data.get('steps', 20)
    
    response = requests.post(
        f"{COMFYUI_SERVER}/prompt",
        json={"prompt": workflow}
    )
    
    return jsonify(response.json())

@app.route('/api/status/<prompt_id>', methods=['GET'])
def get_status(prompt_id):
    """查询任务状态"""
    response = requests.get(
        f"{COMFYUI_SERVER}/history/{prompt_id}"
    )
    return jsonify(response.json())

@app.route('/api/image/<filename>', methods=['GET'])
def get_image(filename):
    """获取生成的图片"""
    response = requests.get(
        f"{COMFYUI_SERVER}/view",
        params={"filename": filename, "type": "output"}
    )
    return response.content

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
```

### FastAPI 封装示例

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import requests
import json

app = FastAPI(title="ComfyUI API Wrapper")
COMFYUI_SERVER = "http://127.0.0.1:8188"

class GenerateRequest(BaseModel):
    prompt: str
    negative_prompt: str = ""
    seed: int = -1
    steps: int = 20
    width: int = 512
    height: int = 512

class GenerateResponse(BaseModel):
    prompt_id: str
    status: str

@app.post("/api/generate", response_model=GenerateResponse)
async def generate(req: GenerateRequest):
    """提交生成任务"""
    with open('templates/basic_txt2img.json') as f:
        workflow = json.load(f)
    
    workflow["6"]["inputs"]["text"] = req.prompt
    workflow["7"]["inputs"]["text"] = req.negative_prompt
    workflow["3"]["inputs"]["seed"] = req.seed
    workflow["3"]["inputs"]["steps"] = req.steps
    workflow["5"]["inputs"]["width"] = req.width
    workflow["5"]["inputs"]["height"] = req.height
    
    response = requests.post(
        f"{COMFYUI_SERVER}/prompt",
        json={"prompt": workflow}
    )
    
    if response.status_code != 200:
        raise HTTPException(status_code=500, detail="ComfyUI error")
    
    data = response.json()
    return GenerateResponse(
        prompt_id=data["prompt_id"],
        status="queued"
    )

@app.get("/api/result/{prompt_id}")
async def get_result(prompt_id: str):
    """获取生成结果"""
    response = requests.get(
        f"{COMFYUI_SERVER}/history/{prompt_id}"
    )
    
    history = response.json()
    if prompt_id not in history:
        return {"status": "pending"}
    
    outputs = history[prompt_id].get("outputs", {})
    images = []
    
    for node_id, output in outputs.items():
        if "images" in output:
            for img in output["images"]:
                images.append({
                    "filename": img["filename"],
                    "url": f"/api/image/{img['filename']}"
                })
    
    return {"status": "completed", "images": images}
```

## 5.5.2 批量生成脚本

### 功能描述

使用 Python 脚本实现批量自动化生成，支持并发生成、进度跟踪和错误处理。

### 基础批量生成

```python
import json
import urllib.request
import time
import random
import os

server_address = "127.0.0.1:8188"

def queue_prompt(workflow):
    data = json.dumps({"prompt": workflow}).encode('utf-8')
    req = urllib.request.Request(
        f"http://{server_address}/prompt",
        data=data,
        headers={'Content-Type': 'application/json'}
    )
    with urllib.request.urlopen(req) as response:
        return json.loads(response.read())

def wait_for_completion(prompt_id, timeout=300):
    start_time = time.time()
    while time.time() - start_time < timeout:
        url = f"http://{server_address}/history/{prompt_id}"
        with urllib.request.urlopen(url) as response:
            history = json.loads(response.read())
            if prompt_id in history:
                return history[prompt_id]
        time.sleep(1)
    raise TimeoutError(f"Timeout waiting for {prompt_id}")

def batch_generate(workflow_path, prompts, output_dir="output"):
    """批量生成图像"""
    os.makedirs(output_dir, exist_ok=True)
    
    with open(workflow_path) as f:
        base_workflow = json.load(f)
    
    results = []
    
    for i, prompt in enumerate(prompts):
        print(f"处理 {i+1}/{len(prompts)}: {prompt[:30]}...")
        
        workflow = json.loads(json.dumps(base_workflow))
        workflow["6"]["inputs"]["text"] = prompt
        workflow["3"]["inputs"]["seed"] = random.randint(0, 2**32 - 1)
        
        result = queue_prompt(workflow)
        history = wait_for_completion(result["prompt_id"])
        
        for node_id, output in history["outputs"].items():
            if "images" in output:
                for img in output["images"]:
                    results.append({
                        "prompt": prompt,
                        "filename": img["filename"],
                        "seed": workflow["3"]["inputs"]["seed"]
                    })
    
    return results

prompts = [
    "a beautiful sunset over mountains",
    "a cat sleeping on a windowsill",
    "a futuristic city at night",
    "a peaceful garden with flowers",
    "an abstract digital artwork"
]

results = batch_generate("workflow_api.json", prompts)
print(f"完成 {len(results)} 张图片生成")
```

### 并发生成

```python
import concurrent.futures
import threading

results_lock = threading.Lock()
results = []

def generate_single(seed, prompt, workflow_path):
    """单个生成任务"""
    with open(workflow_path) as f:
        workflow = json.load(f)
    
    workflow["3"]["inputs"]["seed"] = seed
    workflow["6"]["inputs"]["text"] = prompt
    
    result = queue_prompt(workflow)
    history = wait_for_completion(result["prompt_id"])
    
    with results_lock:
        results.append({
            "seed": seed,
            "prompt": prompt,
            "prompt_id": result["prompt_id"]
        })
    
    return result

def concurrent_batch_generate(prompts, workflow_path, max_workers=3):
    """并发生成"""
    seeds = [random.randint(0, 2**32 - 1) for _ in prompts]
    
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = [
            executor.submit(generate_single, seed, prompt, workflow_path)
            for seed, prompt in zip(seeds, prompts)
        ]
        
        for future in concurrent.futures.as_completed(futures):
            try:
                result = future.result()
                print(f"完成: {result['prompt_id']}")
            except Exception as e:
                print(f"错误: {e}")
    
    return results

prompts = [f"a beautiful landscape {i}" for i in range(10)]
results = concurrent_batch_generate(prompts, "workflow_api.json", max_workers=3)
```

## 5.5.3 数据库记录

### 功能描述

将生成任务和结果存储到数据库，便于追踪、统计和复现。

### 数据库表设计

```sql
CREATE TABLE generations (
    id SERIAL PRIMARY KEY,
    request_id VARCHAR(36) UNIQUE NOT NULL,
    prompt TEXT NOT NULL,
    negative_prompt TEXT,
    seed BIGINT,
    steps INTEGER DEFAULT 20,
    cfg REAL DEFAULT 7.0,
    width INTEGER DEFAULT 512,
    height INTEGER DEFAULT 512,
    sampler VARCHAR(50) DEFAULT 'euler',
    checkpoint VARCHAR(255),
    status VARCHAR(20) DEFAULT 'pending',
    prompt_id VARCHAR(50),
    image_urls JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    error_message TEXT
);

CREATE INDEX idx_generations_status ON generations(status);
CREATE INDEX idx_generations_created ON generations(created_at);
```

### Python 数据库操作

```python
import psycopg2
from psycopg2.extras import Json
import uuid
from datetime import datetime

class GenerationDB:
    def __init__(self, db_url):
        self.conn = psycopg2.connect(db_url)
    
    def create_task(self, prompt, **kwargs):
        """创建生成任务记录"""
        request_id = str(uuid.uuid4())
        
        with self.conn.cursor() as cur:
            cur.execute("""
                INSERT INTO generations 
                (request_id, prompt, negative_prompt, seed, steps, cfg, width, height, status)
                VALUES (%s, %s, %s, %s, %s, %s, %s, %s, 'pending')
                RETURNING id
            """, (
                request_id, prompt,
                kwargs.get('negative_prompt', ''),
                kwargs.get('seed', -1),
                kwargs.get('steps', 20),
                kwargs.get('cfg', 7.0),
                kwargs.get('width', 512),
                kwargs.get('height', 512)
            ))
            task_id = cur.fetchone()[0]
            self.conn.commit()
        
        return request_id, task_id
    
    def update_status(self, request_id, status, prompt_id=None, images=None, error=None):
        """更新任务状态"""
        with self.conn.cursor() as cur:
            if status == 'completed':
                cur.execute("""
                    UPDATE generations 
                    SET status = %s, prompt_id = %s, image_urls = %s, completed_at = %s
                    WHERE request_id = %s
                """, (status, prompt_id, Json(images), datetime.now(), request_id))
            elif status == 'failed':
                cur.execute("""
                    UPDATE generations 
                    SET status = %s, error_message = %s
                    WHERE request_id = %s
                """, (status, error, request_id))
            else:
                cur.execute("""
                    UPDATE generations 
                    SET status = %s, prompt_id = %s
                    WHERE request_id = %s
                """, (status, prompt_id, request_id))
            
            self.conn.commit()
    
    def get_task(self, request_id):
        """获取任务详情"""
        with self.conn.cursor() as cur:
            cur.execute(
                "SELECT * FROM generations WHERE request_id = %s",
                (request_id,)
            )
            return cur.fetchone()
    
    def list_tasks(self, status=None, limit=100):
        """列出任务"""
        with self.conn.cursor() as cur:
            if status:
                cur.execute(
                    "SELECT * FROM generations WHERE status = %s ORDER BY created_at DESC LIMIT %s",
                    (status, limit)
                )
            else:
                cur.execute(
                    "SELECT * FROM generations ORDER BY created_at DESC LIMIT %s",
                    (limit,)
                )
            return cur.fetchall()

db = GenerationDB("postgresql://user:pass@localhost/comfyui_db")
```

```python
import discord
from discord.ext import commands

bot = commands.Bot(command_prefix='!')

@bot.command()
async def generate(ctx, *, prompt):
    """Discord 命令生成图片"""
    await ctx.send(f"正在生成: {prompt[:50]}...")
    
    result = await generate_image_async(prompt)
    
    if result['status'] == 'completed':
        await ctx.send(file=discord.File(result['image_path']))
    else:
        await ctx.send(f"生成失败: {result.get('error', '未知错误')}")

async def generate_image_async(prompt):
    """异步生成图片"""
    import aiohttp
    
    async with aiohttp.ClientSession() as session:
        async with session.post(
            "http://127.0.0.1:5000/api/generate",
            json={"prompt": prompt}
        ) as response:
            data = await response.json()
            return data
```

## 注意事项

- ⚠️ **网络安全**：不要将 ComfyUI 直接暴露在公网，应通过 API 网关访问
- ⚠️ **资源限制**：注意 GPU 资源和显存限制，避免过载
- ⚠️ **并发控制**：ComfyUI 默认串行执行，需要自行管理并发
- 💡 **日志记录**：记录所有生成任务便于排查问题
- 💡 **版本兼容**：注意 API 版本兼容性

## 常见问题

### Q1：如何实现多用户并发？

**A：** 建议：
- 使用消息队列（如 Redis）管理任务
- 部署多个 ComfyUI 实例
- 实现任务调度器分配任务

---

### Q2：集成后响应很慢怎么办？

**A：** 排查方向：
- 检查网络延迟
- 确认 GPU 资源是否充足
- 优化工作流复杂度
- 使用异步处理避免阻塞

---

### Q3：如何保证生成结果的一致性？

**A：** 建议：
- 固定种子值
- 使用相同的模型版本
- 记录完整的生成参数
- 实现参数版本控制

---

### Q4：如何处理大规模批量任务？

**A：** 建议：
- 使用任务队列管理
- 实现断点续传
- 分批处理，避免内存溢出
- 监控进度和资源使用

---

### Q5：如何实现跨平台集成？

**A：** 建议：
- 使用标准 REST API
- 提供 SDK 封装
- 使用 Docker 容器化部署
- 编写详细的 API 文档
