## 案例概述

### 案例目标

本案例将帮助你从零开始搭建一个企业级的 ComfyUI API 服务系统，实现：

1. **高可用架构**：多实例部署，避免单点故障
2. **负载均衡**：请求自动分发到多个 GPU 实例
3. **API 封装**：提供简洁易用的 RESTful API 接口
4. **监控告警**：实时监控系统运行状态

# 定义 ComfyUI 后端服务器组
upstream comfyui_backend {
    # 负载均衡策略：轮询（默认）
    # 可选策略：ip_hash（会话保持）、least_conn（最少连接）
    
    server localhost:8188 weight=1 max_fails=3 fail_timeout=30s;
    server localhost:8189 weight=1 max_fails=3 fail_timeout=30s;
    server localhost:8190 weight=1 max_fails=3 fail_timeout=30s;
    
    # 保持长连接，提高性能
    keepalive 64;
}

# HTTP 服务器配置
server {
    listen 80;
    server_name _;  # 替换为你的域名，如 api.example.com

    # 客户端请求体大小限制（用于上传大图）
    client_max_body_size 50M;

    # 主代理配置
    location / {
        proxy_pass http://comfyui_backend;
        
        # WebSocket 支持（ComfyUI 实时通信需要）
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        
        # 请求头设置
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # 超时设置（图像生成可能需要较长时间）
        proxy_connect_timeout 300s;
        proxy_send_timeout 300s;
        proxy_read_timeout 300s;
        
        # 缓冲设置
        proxy_buffering off;
        proxy_request_buffering off;
    }

    # API 端点专用配置
    location /prompt {
        proxy_pass http://comfyui_backend;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }

    # 健康检查端点
    location /health {
        access_log off;
        return 200 "healthy\n";
        add_header Content-Type text/plain;
    }
}
```

```python
"""
ComfyUI 企业级 API 服务
提供简洁易用的图像生成接口
"""

from fastapi import FastAPI, HTTPException
from fastapi.responses import Response
from pydantic import BaseModel, Field
from typing import Optional, List
import uvicorn
import requests
import uuid
import os
import time
import logging

# 配置日志
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

# 创建 FastAPI 应用
app = FastAPI(
    title="ComfyUI Enterprise API",
    description="企业级图像生成 API 服务",
    version="1.0.0"
)

# 从环境变量读取 ComfyUI 服务器列表
COMFYUI_SERVERS = os.getenv(
    "COMFYUI_SERVERS", 
    "http://localhost:8188,http://localhost:8189,http://localhost:8190"
).split(",")

logger.info(f"ComfyUI servers: {COMFYUI_SERVERS}")


# ==================== 数据模型定义 ====================

class GenerateRequest(BaseModel):
    """图像生成请求模型"""
    prompt: str = Field(..., description="正向提示词，描述你想要生成的图像")
    negative_prompt: str = Field(
        default="low quality, blurry, distorted, ugly, bad anatomy",
        description="负向提示词，描述你不想要的内容"
    )
    width: int = Field(default=512, ge=64, le=2048, description="图像宽度，推荐 512/768/1024")
    height: int = Field(default=512, ge=64, le=2048, description="图像高度，推荐 512/768/1024")
    steps: int = Field(default=20, ge=1, le=100, description="采样步数，越多越精细但越慢")
    cfg: float = Field(default=7.0, ge=1.0, le=20.0, description="CFG 系数，控制提示词遵循程度")
    sampler_name: str = Field(default="euler_ancestral", description="采样器名称")
    scheduler: str = Field(default="normal", description="调度器类型")
    seed: Optional[int] = Field(default=None, description="随机种子，相同种子生成相同图像")
    model_name: str = Field(default="realisticVisionV51_v51VAE.safetensors", description="模型文件名")


class GenerateResponse(BaseModel):
    """生成响应模型"""
    task_id: str = Field(..., description="任务 ID，用于查询结果")
    status: str = Field(..., description="任务状态")
    message: str = Field(..., description="提示信息")


class ResultResponse(BaseModel):
    """结果响应模型"""
    task_id: str
    status: str
    outputs: Optional[dict] = None
    error: Optional[str] = None


# ==================== ComfyUI 客户端 ====================

class ComfyUIClient:
    """ComfyUI 服务器客户端"""
    
    def __init__(self, servers: List[str]):
        self.servers = servers
        self.current_index = 0
    
    def get_next_server(self) -> str:
        """轮询选择下一个服务器"""
        server = self.servers[self.current_index]
        self.current_index = (self.current_index + 1) % len(self.servers)
        logger.info(f"Selected server: {server}")
        return server
    
    def queue_prompt(self, workflow: dict) -> dict:
        """提交工作流到 ComfyUI"""
        server = self.get_next_server()
        url = f"{server}/prompt"
        
        try:
            response = requests.post(
                url,
                json={"prompt": workflow},
                timeout=60
            )
            response.raise_for_status()
            result = response.json()
            logger.info(f"Task queued: {result.get('prompt_id')}")
            return result
        except requests.exceptions.RequestException as e:
            logger.error(f"Failed to queue prompt: {e}")
            raise HTTPException(
                status_code=503, 
                detail=f"ComfyUI 服务暂时不可用: {str(e)}"
            )
    
    def get_history(self, prompt_id: str) -> dict:
        """获取任务历史记录"""
        for server in self.servers:
            try:
                response = requests.get(
                    f"{server}/history/{prompt_id}",
                    timeout=30
                )
                response.raise_for_status()
                history = response.json()
                if prompt_id in history:
                    return history
            except requests.exceptions.RequestException:
                continue
        return {}
    
    def get_image(self, filename: str, subfolder: str = "", img_type: str = "output") -> bytes:
        """获取生成的图像"""
        for server in self.servers:
            try:
                params = {
                    "filename": filename,
                    "subfolder": subfolder,
                    "type": img_type
                }
                response = requests.get(
                    f"{server}/view",
                    params=params,
                    timeout=30
                )
                response.raise_for_status()
                return response.content
            except requests.exceptions.RequestException:
                continue
        raise HTTPException(status_code=404, detail="图像未找到")


# 创建全局客户端实例
client = ComfyUIClient(COMFYUI_SERVERS)


# ==================== 工作流构建 ====================

def create_text2img_workflow(request: GenerateRequest) -> dict:
    """
    根据请求参数构建文生图工作流
    
    工作流结构：
    [模型加载] -> [提示词编码] -> [空 Latent] -> [KSampler 采样] -> [VAE 解码] -> [保存图像]
    """
    # 生成随机种子（如果未指定）
    seed = request.seed if request.seed is not None else int(uuid.uuid4().int % 2**32)
    
    workflow = {
        # KSampler - 核心采样节点
        "3": {
            "class_type": "KSampler",
            "inputs": {
                "seed": seed,
                "steps": request.steps,
                "cfg": request.cfg,
                "sampler_name": request.sampler_name,
                "scheduler": request.scheduler,
                "denoise": 1.0,
                "model": ["4", 0],
                "positive": ["6", 0],
                "negative": ["7", 0],
                "latent_image": ["5", 0]
            }
        },
        # CheckpointLoaderSimple - 模型加载节点
        "4": {
            "class_type": "CheckpointLoaderSimple",
            "inputs": {
                "ckpt_name": request.model_name
            }
        },
        # EmptyLatentImage - 空 Latent 图像节点
        "5": {
            "class_type": "EmptyLatentImage",
            "inputs": {
                "width": request.width,
                "height": request.height,
                "batch_size": 1
            }
        },
        # CLIPTextEncode - 正向提示词节点
        "6": {
            "class_type": "CLIPTextEncode",
            "inputs": {
                "text": request.prompt,
                "clip": ["4", 1]
            }
        },
        # CLIPTextEncode - 负向提示词节点
        "7": {
            "class_type": "CLIPTextEncode",
            "inputs": {
                "text": request.negative_prompt,
                "clip": ["4", 1]
            }
        },
        # VAEDecode - VAE 解码节点
        "8": {
            "class_type": "VAEDecode",
            "inputs": {
                "samples": ["3", 0],
                "vae": ["4", 2]
            }
        },
        # SaveImage - 保存图像节点
        "9": {
            "class_type": "SaveImage",
            "inputs": {
                "filename_prefix": "api_output",
                "images": ["8", 0]
            }
        }
    }
    
    return workflow


# ==================== API 端点定义 ====================

@app.post("/api/v1/generate", response_model=GenerateResponse)
async def generate_image(request: GenerateRequest):
    """
    提交图像生成任务
    
    **使用示例：**
    ```bash
    curl -X POST http://localhost:8000/api/v1/generate \\
      -H "Content-Type: application/json" \\
      -d '{"prompt": "a beautiful sunset over mountains"}'
    ```
    """
    # 构建工作流
    workflow = create_text2img_workflow(request)
    
    # 提交到 ComfyUI
    result = client.queue_prompt(workflow)
    
    return GenerateResponse(
        task_id=result["prompt_id"],
        status="queued",
        message="任务已加入队列，请使用 task_id 查询结果"
    )


@app.get("/api/v1/result/{task_id}", response_model=ResultResponse)
async def get_result(task_id: str):
    """
    查询任务结果
    
    **返回状态说明：**
    - `pending`: 任务等待处理中
    - `processing`: 任务正在处理
    - `completed`: 任务已完成
    - `failed`: 任务失败
    """
    history = client.get_history(task_id)
    
    if not history:
        return ResultResponse(
            task_id=task_id,
            status="pending",
            outputs=None
        )
    
    if task_id not in history:
        return ResultResponse(
            task_id=task_id,
            status="pending",
            outputs=None
        )
    
    outputs = history[task_id].get("outputs", {})
    if not outputs:
        return ResultResponse(
            task_id=task_id,
            status="processing",
            outputs=None
        )
    
    return ResultResponse(
        task_id=task_id,
        status="completed",
        outputs=outputs
    )


@app.get("/api/v1/image/{filename}")
async def get_image(filename: str):
    """
    获取生成的图像文件
    
    **使用示例：**
    ```bash
    curl http://localhost:8000/api/v1/image/api_output_001.png --output image.png
    ```
    """
    image_data = client.get_image(filename)
    return Response(content=image_data, media_type="image/png")


@app.get("/api/v1/health")
async def health_check():
    """健康检查接口"""
    return {
        "status": "healthy",
        "servers": len(client.servers),
        "version": "1.0.0"
    }


@app.get("/api/v1/models")
async def list_models():
    """获取可用模型列表"""
    try:
        server = client.get_next_server()
        response = requests.get(f"{server}/object_info", timeout=30)
        response.raise_for_status()
        return response.json()
    except requests.exceptions.RequestException as e:
        raise HTTPException(
            status_code=503, 
            detail=f"获取模型列表失败: {str(e)}"
        )


# ==================== 启动服务 ====================

if __name__ == "__main__":
    uvicorn.run(
        app,
        host="0.0.0.0",
        port=8000,
        log_level="info"
    )

## 常见问题

### Q1: Docker 容器启动失败，提示 GPU 相关错误

**问题描述：** 运行 `docker compose up -d` 后容器无法启动，日志显示 GPU 相关错误。

**解决方案：**

```bash
# 步骤 1: 检查 NVIDIA 驱动是否正确安装
nvidia-smi
# 如果报错，需要先安装 NVIDIA 驱动

# 步骤 2: 检查 NVIDIA Container Toolkit 是否安装
docker run --rm --gpus all nvidia/cuda:11.8-base nvidia-smi
# 如果报错，重新安装 NVIDIA Container Toolkit

# 步骤 3: 检查 Docker 配置
cat /etc/docker/daemon.json
# 应该包含 "runtimes": {"nvidia": {...}}

# 步骤 4: 重启 Docker
sudo systemctl restart docker
```

```python
from fastapi import Security, HTTPException
from fastapi.security import APIKeyHeader

API_KEY = "your-secret-api-key"
api_key_header = APIKeyHeader(name="X-API-Key")

async def verify_api_key(api_key: str = Security(api_key_header)):
    if api_key != API_KEY:
        raise HTTPException(status_code=403, detail="Invalid API Key")
    return api_key

# 在需要认证的端点添加依赖
@app.post("/api/v1/generate", dependencies=[Depends(verify_api_key)])
async def generate_image(request: GenerateRequest):
    ...
```
