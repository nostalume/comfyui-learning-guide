## 案例概述

### 案例目标

本案例将帮助你搭建一个团队协作平台，实现：

1. **资源共享**：模型、工作流、自定义节点的集中管理
2. **权限管理**：基于角色的访问控制（RBAC）
3. **版本控制**：工作流的版本管理和回滚
4. **审计日志**：操作记录和追溯

### 步骤 2: 配置共享存储（20分钟）

#### 2.1 安装和配置 NFS

**在服务器上执行：**

```bash
# 步骤 1: 安装 NFS 服务器
sudo apt update
sudo apt install nfs-kernel-server -y

# 步骤 2: 创建共享目录（如果尚未创建）
sudo mkdir -p /srv/comfyui-shared

# 步骤 3: 配置 NFS 导出
sudo nano /etc/exports
```

**在 /etc/exports 中添加：**

```
# 格式: 目录 客户端IP(选项)
# 允许所有内网 IP 访问
/srv/comfyui-shared 192.168.1.0/24(rw,sync,no_subtree_check,no_root_squash)

# 或允许特定 IP
/srv/comfyui-shared 192.168.1.100(rw,sync,no_subtree_check,no_root_squash)
/srv/comfyui-shared 192.168.1.101(rw,sync,no_subtree_check,no_root_squash)
```

**应用配置：**

```bash
# 步骤 4: 导出共享目录
sudo exportfs -a

# 步骤 5: 重启 NFS 服务
sudo systemctl restart nfs-kernel-server

# 步骤 6: 验证导出
sudo exportfs -v
# 预期输出: 显示共享目录列表

# 步骤 7: 检查 NFS 服务状态
sudo systemctl status nfs-kernel-server
```

#### 2.2 客户端挂载配置

**在客户端机器上执行：**

```bash
# 步骤 1: 安装 NFS 客户端
sudo apt install nfs-common -y

# 步骤 2: 创建挂载点
sudo mkdir -p /mnt/comfyui-shared

# 步骤 3: 测试挂载
sudo mount -t nfs 服务器IP:/srv/comfyui-shared /mnt/comfyui-shared

# 步骤 4: 验证挂载
df -h | grep comfyui
# 预期输出: 显示挂载信息

# 步骤 5: 配置开机自动挂载
sudo nano /etc/fstab
```

**在 /etc/fstab 中添加：**

```
# NFS 挂载配置
服务器IP:/srv/comfyui-shared /mnt/comfyui-shared nfs defaults 0 0
```

#### 2.3 配置 ComfyUI 使用共享存储

```bash
# 步骤 1: 停止 ComfyUI 服务
# 如果使用 Docker

# 步骤 2: 创建符号链接或修改配置
# 方式 1: 符号链接
ln -s /mnt/comfyui-shared/models ~/ComfyUI/models
ln -s /mnt/comfyui-shared/custom_nodes ~/ComfyUI/custom_nodes

# 方式 2: 修改 Docker Compose 配置
# 将 volumes 改为共享存储路径

# 步骤 3: 重启 ComfyUI 服务
```

```bash
"""
权限管理模块
实现基于角色的访问控制（RBAC）
"""
from fastapi import FastAPI, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer, OAuth2PasswordRequestForm
from jose import JWTError, jwt
from passlib.context import CryptContext
from pydantic import BaseModel
from typing import Optional, List
from enum import Enum
from datetime import datetime, timedelta
import logging

# 配置日志
logging.basicConfig(level=logging.INFO)

# JWT 配置

ALGORITHM = "HS256"

# 密码加密上下文
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

# OAuth2 配置
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")


# ==================== 枚举定义 ====================

class Role(str, Enum):
    """用户角色"""
    ADMIN = "admin"           # 管理员：完全控制
    DEVELOPER = "developer"   # 开发者：读写工作流
    VIEWER = "viewer"         # 查看者：只读


class Permission(str, Enum):
    """权限定义"""
    # 工作流权限
    WORKFLOW_READ = "workflow:read"
    WORKFLOW_WRITE = "workflow:write"
    WORKFLOW_DELETE = "workflow:delete"
    # 模型权限
    MODEL_READ = "model:read"
    MODEL_UPLOAD = "model:upload"
    MODEL_DELETE = "model:delete"
    # 用户管理权限
    USER_MANAGE = "user:manage"
    # 节点管理权限
    NODE_INSTALL = "node:install"
    NODE_DELETE = "node:delete"


# ==================== 角色权限映射 ====================

ROLE_PERMISSIONS = {
    Role.ADMIN: [
        # 工作流权限
        Permission.WORKFLOW_READ, Permission.WORKFLOW_WRITE, Permission.WORKFLOW_DELETE,
        # 模型权限
        Permission.MODEL_READ, Permission.MODEL_UPLOAD, Permission.MODEL_DELETE,
        # 用户管理权限
        Permission.USER_MANAGE,
        # 节点管理权限
        Permission.NODE_INSTALL, Permission.NODE_DELETE
    ],
    Role.DEVELOPER: [
        # 工作流权限

    ],
    Role.VIEWER: [
        # 只读权限
        Permission.WORKFLOW_READ,
        Permission.MODEL_READ
    ]
}

    access_token: str
    token_type: str


class TokenData(BaseModel):
    """Token 数据模型"""
    username: Optional[str] = None
    role: Optional[str] = None

# ==================== 密码和认证函数 ====================

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """验证密码"""
    return pwd_context.verify(plain_password, hashed_password)


def get_password_hash(password: str) -> str:
    """生成密码哈希"""
    return pwd_context.hash(password)


def get_user(db: dict, username: str) -> Optional[UserInDB]:
    """获取用户"""
    if username in db:
        user_dict = db[username]
        return UserInDB(**user_dict)
    return None


def authenticate_user(db: dict, username: str, password: str) -> Optional[UserInDB]:
    """认证用户"""
    user = get_user(db, username)
    if not user:
        return None
    if not verify_password(password, user.hashed_password):
        return None
    return user


def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    """创建访问令牌"""
    to_encode = data.copy()
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=15)
    to_encode.update({"exp": expire})
    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt


async def get_current_user(token: str = Depends(oauth2_scheme)) -> UserInDB:
    """获取当前用户"""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="无法验证凭据",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        role: str = payload.get("role")
        if username is None:
            raise credentials_exception
        token_data = TokenData(username=username, role=role)
    except JWTError:
        raise credentials_exception
    
    user = get_user(fake_users_db, username=token_data.username)
    if user is None:
        raise credentials_exception
    if not user.is_active:
        raise HTTPException(status_code=400, detail="用户已被禁用")
    return user


def check_permission(user: UserInDB, required_permission: Permission) -> bool:
    """检查用户是否拥有指定权限"""
    user_permissions = ROLE_PERMISSIONS.get(user.role, [])
    return required_permission in user_permissions


def require_permission(permission: Permission):
    """权限检查装饰器"""
    async def permission_checker(current_user: UserInDB = Depends(get_current_user)):
        if not check_permission(current_user, permission):
            logger.warning(f"用户 {current_user.username} 尝试访问无权限的资源: {permission}")
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"权限不足，需要 {permission.value} 权限"
            )
        return current_user
    return permission_checker
```

### 步骤 4: 实现工作流版本控制（20分钟）

#### 4.1 创建版本控制模块

```bash
cat > workflow_version.py << 'EOF'
"""
工作流版本控制模块
使用 Git 管理工作流版本
"""
import git
import json
import os
from datetime import datetime
from pathlib import Path
from typing import Optional, List, Dict, Any
import logging

logger = logging.getLogger(__name__)


class WorkflowVersionControl:
    """工作流版本管理器"""
    
    def __init__(self, repo_path: str = "./workflows-repo"):
        """
        初始化版本控制器
        
        Args:
            repo_path: Git 仓库路径
        """
        self.repo_path = Path(repo_path)
        self._init_repository()
    
    def _init_repository(self):
        """初始化 Git 仓库"""
        if not (self.repo_path / ".git").exists():
            # 创建目录
            self.repo_path.mkdir(parents=True, exist_ok=True)
            
            # 初始化 Git 仓库
            self.repo = git.Repo.init(str(self.repo_path))
            
            # 创建初始结构
            workflows_dir = self.repo_path / "workflows"
            workflows_dir.mkdir(exist_ok=True)
            
            # 创建 .gitkeep 文件
            (workflows_dir / ".gitkeep").write_text("")
            
            # 初始提交
            self.repo.index.add(["workflows/.gitkeep"])
            self.repo.index.commit("初始化工作流仓库")
            
            logger.info(f"已初始化工作流仓库: {self.repo_path}")
        else:
            self.repo = git.Repo(str(self.repo_path))
            logger.info(f"已加载工作流仓库: {self.repo_path}")
    
    def save_workflow(
        self, 
        workflow_id: str, 
        workflow_data: Dict[str, Any],
        user_id: str,
        message: str = "更新工作流"
    ) -> str:
        """
        保存工作流并创建版本
        
        Args:
            workflow_id: 工作流 ID
            workflow_data: 工作流数据
            user_id: 用户 ID
            message: 提交消息
            
        Returns:
            提交哈希
        """
        # 构建文件路径
        workflow_path = self.repo_path / "workflows" / f"{workflow_id}.json"
        workflow_path.parent.mkdir(parents=True, exist_ok=True)
        
        # 准备保存的数据
        save_data = {
            "id": workflow_id,
            "data": workflow_data,
            "author": user_id,
            "updated_at": datetime.now().isoformat()
        }
        
        # 写入文件
        with open(workflow_path, 'w', encoding='utf-8') as f:
            json.dump(save_data, f, indent=2, ensure_ascii=False)
        
        # 提交到 Git
        self.repo.index.add([str(workflow_path.relative_to(self.repo_path))])
        commit = self.repo.index.commit(f"[{workflow_id}] {message}")
        
        logger.info(f"已保存工作流: {workflow_id}, 提交: {commit.hexsha[:8]}")
        return commit.hexsha
    
    def get_workflow(
        self, 
        workflow_id: str, 
        version_hash: Optional[str] = None
    ) -> Optional[Dict[str, Any]]:
        """
        获取工作流
        
        Args:
            workflow_id: 工作流 ID
            version_hash: 版本哈希（可选，默认获取最新版本）
            
        Returns:
            工作流数据或 None
        """
        workflow_path = f"workflows/{workflow_id}.json"
        
        if version_hash:
            # 获取指定版本
            try:
                content = self.repo.git.show(f"{version_hash}:{workflow_path}")
                return json.loads(content)
            except git.GitCommandError as e:
                logger.error(f"获取工作流版本失败: {e}")
                return None
        
        # 获取最新版本
        full_path = self.repo_path / workflow_path
        if not full_path.exists():
            return None
        
        with open(full_path, encoding='utf-8') as f:
            return json.load(f)
    
    def list_versions(self, workflow_id: str, limit: int = 20) -> List[Dict[str, Any]]:
        """
        列出工作流的所有版本
        
        Args:
            workflow_id: 工作流 ID
            limit: 最大返回数量
            
        Returns:
            版本列表
        """
        workflow_path = f"workflows/{workflow_id}.json"
        versions = []
        
        try:
            commits = list(self.repo.iter_commits(
                paths=workflow_path, 
                max_count=limit
            ))
            
            for commit in commits:
                versions.append({
                    "hash": commit.hexsha[:8],
                    "full_hash": commit.hexsha,
                    "message": commit.message.strip(),
                    "author": commit.author.name,
                    "email": commit.author.email,
                    "date": commit.committed_datetime.isoformat()
                })
        except git.GitCommandError as e:
            logger.error(f"获取版本列表失败: {e}")
        
        return versions
    
    def restore_version(self, workflow_id: str, version_hash: str, user_id: str) -> str:
        """
        恢复到指定版本
        
        Args:
            workflow_id: 工作流 ID
            version_hash: 版本哈希
            user_id: 操作用户 ID
            
        Returns:
            新提交哈希
        """
        workflow_path = f"workflows/{workflow_id}.json"
        
        # 从指定版本获取内容
        content = self.repo.git.show(f"{version_hash}:{workflow_path}")
        
        # 写回文件
        full_path = self.repo_path / workflow_path
        with open(full_path, 'w', encoding='utf-8') as f:
            f.write(content)
        
        # 提交恢复操作
        self.repo.index.add([workflow_path])
        commit = self.repo.index.commit(
            f"[{workflow_id}] 恢复到版本 {version_hash[:8]}"
        )
        
        logger.info(f"已恢复工作流 {workflow_id} 到版本 {version_hash[:8]}")
        return commit.hexsha
    
    def delete_workflow(self, workflow_id: str, user_id: str) -> bool:
        """
        删除工作流
        
        Args:
            workflow_id: 工作流 ID
            user_id: 操作用户 ID
            
        Returns:
            是否成功
        """
        workflow_path = self.repo_path / "workflows" / f"{workflow_id}.json"
        
        if not workflow_path.exists():
            return False
        
        # 删除文件
        workflow_path.unlink()
        
        # 提交删除
        self.repo.index.remove([f"workflows/{workflow_id}.json"])
        self.repo.index.commit(f"[{workflow_id}] 删除工作流")
        
        logger.info(f"已删除工作流: {workflow_id}")
        return True
    
    def list_workflows(self) -> List[Dict[str, Any]]:
        """
        列出所有工作流
        
        Returns:
            工作流列表
        """
        workflows_dir = self.repo_path / "workflows"
        workflows = []
        
        for json_file in workflows_dir.glob("*.json"):
            try:
                with open(json_file, encoding='utf-8') as f:
                    data = json.load(f)
                    workflows.append({
                        "id": data.get("id"),
                        "author": data.get("author"),
                        "updated_at": data.get("updated_at")
                    })
            except Exception as e:
                logger.error(f"读取工作流文件失败: {e}")
        
        # 按更新时间排序
        workflows.sort(key=lambda x: x.get("updated_at", ""), reverse=True)
        return workflows
```

```bash
"""
审计日志模块
记录用户操作日志
"""
import json
from datetime import datetime
from pathlib import Path
from typing import Dict, Any, List, Optional
import logging

logger = logging.getLogger(__name__)


class AuditLogger:
    """审计日志记录器"""
    
    def __init__(self, log_dir: str = "./audit-logs"):
        """
        初始化审计日志
        
        Args:
            log_dir: 日志目录
        """
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(parents=True, exist_ok=True)
        logger.info(f"审计日志目录: {self.log_dir}")
    
    def log(
        self,
        user_id: str,
        action: str,
        resource_type: str,
        resource_id: str,
        details: Optional[Dict[str, Any]] = None,
        ip_address: Optional[str] = None
    ):
        """
        记录审计日志
        
        Args:
            user_id: 用户 ID
            action: 操作类型 (create/update/delete/read)
            resource_type: 资源类型 (workflow/model/node)
            resource_id: 资源 ID
            details: 详细信息
            ip_address: 客户端 IP
        """
        log_entry = {
            "timestamp": datetime.now().isoformat(),
            "user_id": user_id,
            "action": action,
            "resource_type": resource_type,
            "resource_id": resource_id,
            "details": details or {},
            "ip_address": ip_address
        }
        
        # 按日期创建日志文件
        date_str = datetime.now().strftime("%Y-%m-%d")
        log_file = self.log_dir / f"{date_str}.log"
        
        # 写入日志
        with open(log_file, 'a', encoding='utf-8') as f:
            f.write(json.dumps(log_entry, ensure_ascii=False) + "\n")
        
        logger.info(f"审计日志: {user_id} {action} {resource_type}/{resource_id}")
    
    def get_logs(
        self,
        user_id: Optional[str] = None,
        action: Optional[str] = None,
        resource_type: Optional[str] = None,
        date: Optional[str] = None,
        limit: int = 100
    ) -> List[Dict[str, Any]]:
        """
        查询审计日志
        
        Args:
            user_id: 用户 ID 过滤
            action: 操作类型过滤
            resource_type: 资源类型过滤
            date: 日期过滤 (YYYY-MM-DD)
            limit: 最大返回数量
            
        Returns:
            日志列表
        """
        logs = []
        
        # 确定日志文件
        if date:
            log_files = [self.log_dir / f"{date}.log"]
        else:
            log_files = sorted(self.log_dir.glob("*.log"), reverse=True)
        
        for log_file in log_files:
            if not log_file.exists():
                continue
            
            with open(log_file, encoding='utf-8') as f:
                for line in f:
                    try:
                        entry = json.loads(line.strip())
                        
                        # 应用过滤条件
                        if user_id and entry.get("user_id") != user_id:
                            continue
                        if action and entry.get("action") != action:
                            continue
                        if resource_type and entry.get("resource_type") != resource_type:
                            continue
                        
                        logs.append(entry)
                        
                        if len(logs) >= limit:
                            return logs
                    except json.JSONDecodeError:
                        continue
        
        return logs
    
    def get_user_activity(self, user_id: str, days: int = 7) -> Dict[str, Any]:
        """
        获取用户活动统计
        
        Args:
            user_id: 用户 ID
            days: 统计天数
            
        Returns:
            活动统计
        """
        from datetime import datetime, timedelta
        
        activities = {
            "total_actions": 0,
            "by_action": {},
            "by_resource_type": {},
            "recent_logs": []
        }
        
        # 获取最近几天的日志
        for i in range(days):
            date = (datetime.now() - timedelta(days=i)).strftime("%Y-%m-%d")
            logs = self.get_logs(user_id=user_id, date=date)
            
            for log in logs:
                activities["total_actions"] += 1
                
                # 按操作类型统计
                action = log.get("action", "unknown")
                activities["by_action"][action] = activities["by_action"].get(action, 0) + 1
                
                # 按资源类型统计
                resource_type = log.get("resource_type", "unknown")
                activities["by_resource_type"][resource_type] = activities["by_resource_type"].get(resource_type, 0) + 1
            
            activities["recent_logs"].extend(logs[:10])
        
        return activities


# 全局审计日志实例
audit_logger = AuditLogger()
```
