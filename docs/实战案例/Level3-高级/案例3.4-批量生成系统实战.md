### 步骤 1: 设计批量生成策略

#### 1.1 常见批量模式对比

| 模式 | 说明 | 适用场景 | 复杂度 | 示例 |
|-----|------|---------|-------|------|
| **提示词矩阵** | 多个提示词维度组合 | 风格探索、创意生成 | 中 | 主体×风格×场景 |
| **参数遍历** | 单个或多个参数值变化 | 参数优化、效果测试 | 低-中 | CFG、steps、seed |
| **种子遍历** | 相同参数不同种子 | 结果筛选、多样化 | 低 | 50个不同种子 |
| **LoRA 切换** | 多个 LoRA 模型测试 | 模型选择、风格混合 | 中 | LoRA A/B/C 对比 |
| **混合模式** | 多种模式组合 | 复杂测试场景 | 高 | 提示词×CFG×种子 |

#### 1.2 选择建议

| 需求 | 推荐模式 | 说明 |
|-----|---------|------|
| 探索不同风格 | 提示词矩阵 | 系统化测试各种组合 |
| 找最佳参数 | 参数遍历 | 科学化参数优化 |
| 筛选最佳结果 | 种子遍历 | 大量生成后人工筛选 |
| 测试模型效果 | LoRA 切换 | 对比不同模型 |
| 复杂测试 | 混合模式 | 综合多种因素 |

### 步骤 2: 创建批量生成器类

创建文件 `batch_generator.py`：

```python
import json
import os
import time
import random
import itertools
from datetime import datetime
from typing import Dict, Any, List, Optional
from tqdm import tqdm
import requests

class BatchGenerator:
    """
    批量生成器类
    支持多种批量模式：提示词矩阵、参数遍历、种子遍历等
    """
    
    def __init__(self, server: str = "127.0.0.1", port: int = 8188):
        """
        初始化批量生成器
        
        Args:
            server: ComfyUI 服务器地址
            port: ComfyUI 服务端口
        """
        self.server = server
        self.port = port
        self.base_url = f"http://{server}:{port}"
        self.workflow = None
        self.output_base_dir = "batch_output"
    
    def load_workflow(self, workflow_path: str) -> None:
        """
        加载工作流模板
        
        Args:
            workflow_path: 工作流 JSON 文件路径
        """
        with open(workflow_path, "r", encoding="utf-8") as f:
            self.workflow = json.load(f)
        print(f"工作流已加载: {workflow_path}")
    
    def set_output_dir(self, output_dir: str) -> None:
        """设置输出目录"""
        self.output_base_dir = output_dir
        os.makedirs(output_dir, exist_ok=True)
    
    def _queue_prompt(self, workflow: Dict[str, Any]) -> str:
        """提交工作流并返回 prompt_id"""
        url = f"{self.base_url}/prompt"
        response = requests.post(url, json={"prompt": workflow})
        response.raise_for_status()
        return response.json()["prompt_id"]
    
    def _wait_for_completion(self, prompt_id: str, timeout: int = 300) -> Dict[str, Any]:
        """等待工作流完成"""
        start = time.time()
        while time.time() - start < timeout:
            response = requests.get(f"{self.base_url}/history/{prompt_id}")
            history = response.json()
            if prompt_id in history:
                status = history[prompt_id].get("status", {})
                if status.get("completed", False):
                    return history[prompt_id]
                if status.get("error", False):
                    raise RuntimeError(f"生成失败: {status.get('error_message')}")
            time.sleep(1)
        raise TimeoutError("生成超时")
    
    def _download_images(self, history: Dict[str, Any], output_dir: str) -> List[str]:
        """下载生成的图像"""
        saved_paths = []
        os.makedirs(output_dir, exist_ok=True)
        
        for node_id, output in history.get("outputs", {}).items():
            if "images" in output:
                for img_info in output["images"]:
                    filename = img_info["filename"]
                    subfolder = img_info.get("subfolder", "")
                    
                    url = f"{self.base_url}/view"
                    params = {"filename": filename, "subfolder": subfolder, "type": "output"}
                    response = requests.get(url, params=params)
                    
                    save_path = os.path.join(output_dir, filename)
                    with open(save_path, "wb") as f:
                        f.write(response.content)
                    saved_paths.append(save_path)
        
        return saved_paths
    
    def _find_node_by_type(self, workflow: Dict[str, Any], class_type: str) -> Optional[str]:
        """根据类型查找节点 ID"""
        for node_id, node_data in workflow.items():
            if node_data.get("class_type") == class_type:
                return node_id
        return None
    
    def prompt_matrix(self, prompt_components: Dict[str, List[str]], 
                      output_prefix: str = "prompt_matrix") -> None:
        """
        提示词矩阵批量生成
        
        Args:
            prompt_components: 提示词组件字典
            output_prefix: 输出目录前缀
        """
        if self.workflow is None:
            raise ValueError("请先加载工作流")
        
        keys = list(prompt_components.keys())
        values = list(prompt_components.values())
        combinations = list(itertools.product(*values))
        
        print(f"开始提示词矩阵生成，共 {len(combinations)} 种组合")
        
        batch_dir = os.path.join(
            self.output_base_dir, 
            f"{output_prefix}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        )
        os.makedirs(batch_dir, exist_ok=True)
        
        for idx, combo in enumerate(tqdm(combinations, desc="生成进度")):
            full_prompt = ", ".join(combo)
            
            workflow = json.loads(json.dumps(self.workflow))
            
            node_id = self._find_node_by_type(workflow, "CLIPTextEncode")
            if node_id:
                workflow[node_id]["inputs"]["text"] = full_prompt
            
            try:
                prompt_id = self._queue_prompt(workflow)
                history = self._wait_for_completion(prompt_id)
                
                combo_dir = os.path.join(batch_dir, f"{idx+1:03d}_{'-'.join(combo)}")
                self._download_images(history, combo_dir)
                
                metadata = {
                    "prompt": full_prompt,
                    "components": dict(zip(keys, combo)),
                    "prompt_id": prompt_id,
                    "timestamp": datetime.now().isoformat()
                }
                with open(os.path.join(combo_dir, "metadata.json"), "w", encoding="utf-8") as f:
                    json.dump(metadata, f, indent=2, ensure_ascii=False)
                
            except Exception as e:
                print(f"\n组合 {idx+1} 失败: {e}")
                continue
        
        print(f"\n提示词矩阵生成完成，结果保存在: {batch_dir}")
    
    def param_sweep(self, param_spec: Dict[str, List[Any]],
                   output_prefix: str = "param_sweep") -> None:
        """
        参数遍历批量生成
        
        Args:
            param_spec: 参数规格字典，如 {"3/cfg": [5, 7, 10]}
            output_prefix: 输出目录前缀
        """
        if self.workflow is None:
            raise ValueError("请先加载工作流")
        
        keys = list(param_spec.keys())
        values = list(param_spec.values())
        combinations = list(itertools.product(*values))
        
        print(f"开始参数遍历，共 {len(combinations)} 种组合")
        
        batch_dir = os.path.join(
            self.output_base_dir,
            f"{output_prefix}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        )
        os.makedirs(batch_dir, exist_ok=True)
        
        for idx, combo in enumerate(tqdm(combinations, desc="生成进度")):
            workflow = json.loads(json.dumps(self.workflow))
            
            param_desc = []
            for key, value in zip(keys, combo):
                node_id, param_name = key.split("/")
                if node_id in workflow:
                    workflow[node_id]["inputs"][param_name] = value
                    param_desc.append(f"{param_name}={value}")
            
            try:
                prompt_id = self._queue_prompt(workflow)
                history = self._wait_for_completion(prompt_id)
                
                combo_dir = os.path.join(batch_dir, f"{idx+1:03d}_{'-'.join(param_desc)}")
                self._download_images(history, combo_dir)
                
                metadata = {
                    "params": dict(zip(keys, combo)),
                    "prompt_id": prompt_id,
                    "timestamp": datetime.now().isoformat()
                }
                with open(os.path.join(combo_dir, "metadata.json"), "w", encoding="utf-8") as f:
                    json.dump(metadata, f, indent=2)
                
            except Exception as e:
                print(f"\n组合 {idx+1} 失败: {e}")
                continue
        
        print(f"\n参数遍历完成，结果保存在: {batch_dir}")
    
    def seed_batch(self, count: int = 10, output_prefix: str = "seed_batch") -> None:
        """
        种子遍历批量生成
        
        Args:
            count: 生成数量
            output_prefix: 输出目录前缀
        """
        if self.workflow is None:
            raise ValueError("请先加载工作流")
        
        print(f"开始种子遍历，共生成 {count} 张图像")
        
        batch_dir = os.path.join(
            self.output_base_dir,
            f"{output_prefix}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        )
        os.makedirs(batch_dir, exist_ok=True)
        
        for idx in tqdm(range(count), desc="生成进度"):
            workflow = json.loads(json.dumps(self.workflow))
            
            seed = random.randint(0, 2**32 - 1)
            node_id = self._find_node_by_type(workflow, "KSampler")
            if node_id:
                workflow[node_id]["inputs"]["seed"] = seed
            
            try:
                prompt_id = self._queue_prompt(workflow)
                history = self._wait_for_completion(prompt_id)
                
                seed_dir = os.path.join(batch_dir, f"{idx+1:03d}_seed_{seed}")
                self._download_images(history, seed_dir)
                
                metadata = {
                    "seed": seed,
                    "prompt_id": prompt_id,
                    "timestamp": datetime.now().isoformat()
                }
                with open(os.path.join(seed_dir, "metadata.json"), "w") as f:
                    json.dump(metadata, f, indent=2)
                
            except Exception as e:
                print(f"\n第 {idx+1} 张失败: {e}")
                continue
        
        print(f"\n种子遍历完成，结果保存在: {batch_dir}")

# 创建批量生成器实例
generator = BatchGenerator(server="127.0.0.1", port=8188)
print("批量生成器已创建")
```

## ❌ 常见问题与解决方案

### 问题 1: 批量生成中断

**症状：** 批量生成过程中断，部分任务未完成

**可能原因：**
- 网络连接中断
- ComfyUI 服务崩溃
- 显存不足

**解决方案：**

1. **添加断点续传：**
   ```python
   # 记录已完成的任务
   completed_file = "completed_tasks.json"
   
   def save_progress(completed_list):
       with open(completed_file, "w") as f:
           json.dump(completed_list, f)
   
   def load_progress():
       if os.path.exists(completed_file):
           with open(completed_file, "r") as f:
               return json.load(f)
       return []
   ```

2. **添加错误重试：**
   ```python
   def retry_on_error(func, max_retries=3):
       for i in range(max_retries):
           try:
               return func()
           except Exception as e:
               if i == max_retries - 1:
                   raise
               time.sleep(5)
   ```

### 问题 2: 输出目录混乱

**症状：** 生成的文件难以管理和查找

**解决方案：**

1. **使用时间戳命名：**
   ```python
   batch_dir = f"batch_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
   ```

2. **保存详细元数据：**
   ```python
   metadata = {
       "batch_id": batch_id,
       "total_count": total,
       "current_index": idx,
       "parameters": params,
       "timestamp": datetime.now().isoformat()
   }
   ```

### 问题 3: 内存占用过高

**症状：** 批量生成时内存持续增长

**解决方案：**

1. **及时清理变量：**
   ```python
   import gc
   gc.collect()
   ```

2. **分批处理：**
   ```python
   def batch_process(items, batch_size=10):
       for i in range(0, len(items), batch_size):
           batch = items[i:i+batch_size]
           process_batch(batch)
           gc.collect()
   ```

### 问题 4: 进度无法追踪

**症状：** 不知道批量生成进度

**解决方案：**

使用 tqdm 显示进度条：
```python
from tqdm import tqdm

for item in tqdm(items, desc="生成进度"):
    process_item(item)
```

### 问题 5: 参数 ID 错误

**症状：** 修改参数时提示节点不存在

**解决方案：**

1. **打印所有节点：**
   ```python
   for node_id, node_data in workflow.items():
       print(f"ID: {node_id}, Type: {node_data.get('class_type')}")
   ```

2. **使用动态查找：**
   ```python
   def find_node_by_type(workflow, class_type):
       for node_id, node_data in workflow.items():
           if node_data.get("class_type") == class_type:
               return node_id
       return None
   ```
