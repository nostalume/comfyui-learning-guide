```python
import bpy
import os

def setup_depth_render(scene_name: str = "Scene"):
    """
    配置 Blender 渲染深度图
    
    Args:
        scene_name: 场景名称
    """
    scene = bpy.data.scenes[scene_name]
    
    # 启用合成
    scene.render.use_compositing = True
    scene.node_tree.nodes.clear()
    
    # 创建节点
    render_layers = scene.node_tree.nodes.new('CompositorNodeRLayers')
    normalize = scene.node_tree.nodes.new('CompositorNodeNormalize')
    file_output = scene.node_tree.nodes.new('CompositorNodeOutputFile')
    
    # 设置输出路径
    output_dir = "blender_output"
    os.makedirs(output_dir, exist_ok=True)
    file_output.base_path = output_dir
    file_output.file_slots[0].path = "depth_"
    
    # 连接节点
    scene.node_tree.links.new(
        render_layers.outputs['Depth'],
        normalize.inputs['Value']
    )
    scene.node_tree.links.new(
        normalize.outputs['Value'],
        file_output.inputs[0]
    )
    
    # 设置渲染格式
    scene.render.image_settings.file_format = 'PNG'
    scene.render.resolution_x = 1024
    scene.render.resolution_y = 1024
    
    print("深度图渲染配置完成")

def render_depth_map(output_path: str = "depth.png") -> str:
    """
    渲染深度图
    
    Args:
        output_path: 输出路径
    
    Returns:
        深度图文件路径
    """
    scene = bpy.context.scene
    
    # 渲染
    bpy.ops.render.render(write_still=True)
    
    # 移动文件到指定位置
    import shutil
    temp_path = os.path.join("blender_output", "depth_0001.png")
    if os.path.exists(temp_path):
        shutil.move(temp_path, output_path)
        print(f"深度图已保存: {output_path}")
        return output_path
    else:
        raise FileNotFoundError("渲染文件未生成")

# 使用示例
if __name__ == "__main__":
    setup_depth_render()
    render_depth_map("comfyui_input/depth.png")
```

### 步骤 3: FFmpeg 视频处理集成

#### 3.1 视频帧提取

```python
import subprocess
import os

def extract_frames(video_path: str, output_dir: str = "frames") -> list:
    """
    从视频中提取帧
    
    Args:
        video_path: 视频文件路径
        output_dir: 帧输出目录
    
    Returns:
        帧文件路径列表
    """
    os.makedirs(output_dir, exist_ok=True)
    
    # FFmpeg 命令
    cmd = [
        "ffmpeg",
        "-i", video_path,
        "-vf", "fps=10",  # 每秒提取 10 帧
        "-q:v", "2",      # 高质量
        f"{output_dir}/frame_%04d.png"
    ]
    
    subprocess.run(cmd, check=True)
    
    # 获取所有帧文件
    frames = sorted([f for f in os.listdir(output_dir) if f.endswith(".png")])
    frame_paths = [os.path.join(output_dir, f) for f in frames]
    
    print(f"已提取 {len(frame_paths)} 帧")
    return frame_paths
```

#### 3.2 帧合成为视频

```python
def frames_to_video(frame_dir: str, output_path: str, fps: int = 10) -> None:
    """
    将帧合成为视频
    
    Args:
        frame_dir: 帧目录
        output_path: 输出视频路径
        fps: 帧率
    """
    cmd = [
        "ffmpeg",
        "-framerate", str(fps),
        "-i", f"{frame_dir}/frame_%04d.png",
        "-c:v", "libx264",
        "-r", str(fps),
        "-pix_fmt", "yuv420p",
        output_path
    ]
    
    subprocess.run(cmd, check=True)
    print(f"视频已生成: {output_path}")
```

    frames_to_video(output_frames, output_video, fps=10)

## ❌ 常见问题与解决方案

### 问题 1: Blender 脚本执行失败

**症状：** Blender Python 脚本无法正常执行

**可能原因：**
- 脚本路径错误
- Python 环境不匹配
- 权限问题

**解决方案：**

1. **检查脚本路径：**
   ```bash
   blender --background scene.blend --python script.py
   ```

2. **检查 Blender Python 环境：**
   ```python
   import sys
   print(sys.executable)
   print(sys.path)
   ```

### 问题 2: FFmpeg 命令找不到

**症状：** `ffmpeg: command not found`

**解决方案：**

1. **检查 FFmpeg 安装：**
   ```bash
   ffmpeg -version
   ```

2. **添加到环境变量：**
   - Windows: 将 FFmpeg bin 目录添加到 PATH
   - Linux/Mac: 使用包管理器安装

### 问题 3: 文件监控不工作

**症状：** watchdog 监控不到文件变化

**可能原因：**
- 监控目录不存在
- 权限问题
- 平台兼容性

**解决方案：**

1. **确保目录存在：**
   ```python
   os.makedirs("watch_folder", exist_ok=True)
   ```

2. **检查权限：**
   ```python
   # Linux/Mac
   os.chmod("watch_folder", 0o755)
   ```

### 问题 4: 集成工作流中断

**症状：** 自动化流程中途停止

**解决方案：**

1. **添加错误处理：**
   ```python
   try:
       result = process_step()
   except Exception as e:
       print(f"错误: {e}")
       # 记录错误并继续
   ```

2. **添加重试机制：**
   ```python
   def retry(func, max_retries=3):
       for i in range(max_retries):
           try:
               return func()
           except Exception as e:
               if i == max_retries - 1:
                   raise
               time.sleep(5)
   ```

### 问题 5: 跨平台兼容性问题

**症状：** 代码在不同操作系统上表现不同

**解决方案：**

1. **使用跨平台路径：**
   ```python
   import os
   path = os.path.join("folder", "file.png")
   ```

2. **检测操作系统：**
   ```python
   import platform
   system = platform.system()  # 'Windows', 'Linux', 'Darwin'
   ```
